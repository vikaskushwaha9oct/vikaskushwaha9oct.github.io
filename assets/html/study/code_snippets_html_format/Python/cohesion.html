class CoordinateAndStringUtils:
    def __init__(self, x: int, y: int):
        self.x = x
        self.y = y
        
    @staticmethod
    def reverse_string(string: str) -> str:
        sb: List[str] = []
        i: int = len(string) - 1
        while i >= 0:
            sb.append(string[i])
            i -= 1
        return "".join(sb)
        
    def get_x(self) -> int:
        return self.x
        
    def get_y(self) -> int:
        return self.y
                
def test() -> str:
    return CoordinateAndStringUtils.reverse_string("abcd")
